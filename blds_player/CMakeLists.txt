#-------------------------------------------------------------------------------
# Embedded BLDS files. Only the six exact names are accepted; any that are
# present are embedded, missing ones are reported as "No BLDS file" at runtime.
#
# The bytes are pulled in verbatim by the assembler's .incbin directive (no
# code generator, no hex conversion), via a generated C++ file.
#-------------------------------------------------------------------------------
set(BLDS_RATES 44100 48000 88200 96000 176400 192000)
set(BLDS_DIR "${CMAKE_CURRENT_SOURCE_DIR}/blds_files")
set(BLDS_EMBEDDED_FILES "")
set(BLDS_ASM "")
set(BLDS_DECLS "")
set(BLDS_ENTRIES "")
set(BLDS_COUNT 0)
set(BLDS_TOTAL_BYTES 0)
set(BLDS_SIZE_REPORT "")

# Everything embedded is also decoded into RAM in every plugin instance, and a
# Mach-O object can't hold more than a few GB, so the total is capped.
set(BLDS_MAX_EMBED_MB 256 CACHE STRING "Maximum total size of the embedded BLDS files in MB")

# Re-run configure if a file appears or disappears.
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${BLDS_DIR}")

foreach(rate IN LISTS BLDS_RATES)
    set(f "${BLDS_DIR}/BLDS_${rate}.wav")

    if(EXISTS "${f}" AND NOT IS_DIRECTORY "${f}")
        get_filename_component(f "${f}" REALPATH)

        if(f MATCHES "[\"\\\n]")
            message(FATAL_ERROR "BLDS: the path '${f}' contains a quote or backslash; move the project to a simpler path.")
        endif()

        file(SIZE "${f}" size)
        math(EXPR BLDS_TOTAL_BYTES "${BLDS_TOTAL_BYTES} + ${size}")
        math(EXPR seconds "(${size} - 44) / 3 / ${rate}")
        math(EXPR mb "${size} / 1048576")
        string(APPEND BLDS_SIZE_REPORT "    BLDS_${rate}.wav  ${mb} MB  (~${seconds} s)\n")
        file(SHA256 "${f}" sha)
        set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${f}")
        list(APPEND BLDS_EMBEDDED_FILES "${f}")
        math(EXPR BLDS_COUNT "${BLDS_COUNT} + 1")
        message(STATUS "BLDS: embedding BLDS_${rate}.wav (${size} bytes, sha256 ${sha})")

        set(sym "blds_wav_${rate}")
        string(APPEND BLDS_ASM
            "__asm__ (BLDS_SECTION\n"
            "         BLDS_HIDDEN (\"${sym}\") \"\\n\"\n"
            "         BLDS_HIDDEN (\"${sym}_end\") \"\\n\"\n"
            "         \".p2align 4\\n\"\n"
            "         BLDS_LABEL (\"${sym}\") \":\\n\"\n"
            "         \".incbin \\\"${f}\\\"\\n\"\n"
            "         BLDS_LABEL (\"${sym}_end\") \":\\n\"\n"
            "         \".byte 0\\n\"\n"
            "         \".text\\n\");\n\n")
        string(APPEND BLDS_DECLS
            "extern \"C\" const unsigned char ${sym}[];\n"
            "extern \"C\" const unsigned char ${sym}_end[];\n")
        string(APPEND BLDS_ENTRIES
            "    { ${rate}, \"BLDS_${rate}.wav\", ${sym}, ${sym}_end, ${size}u },\n")
    else()
        message(WARNING "BLDS: blds_files/BLDS_${rate}.wav not found - ${rate} Hz will report 'No BLDS file'")
    endif()
endforeach()

math(EXPR BLDS_TOTAL_MB "${BLDS_TOTAL_BYTES} / 1048576")
if(BLDS_TOTAL_MB GREATER BLDS_MAX_EMBED_MB)
    message(FATAL_ERROR
        "BLDS: the files in blds_files/ total ${BLDS_TOTAL_MB} MB, more than the ${BLDS_MAX_EMBED_MB} MB that can be embedded:\n"
        "${BLDS_SIZE_REPORT}"
        "If BLDS Generator lets you choose a shorter duration, generate shorter files (1 minute at all six "
        "rates is about 115 MB), or keep only the rates you actually use, then delete the build folder and "
        "run the script again.")
endif()

if(BLDS_COUNT EQUAL 0)
    set(BLDS_ENTRIES "    { 0, \"\", nullptr, nullptr, 0u }  // placeholder: nothing embedded\n")
endif()

set(BLDS_EMBED_CPP "${CMAKE_CURRENT_BINARY_DIR}/blds_embedded/BldsEmbeddedData.cpp")
file(CONFIGURE OUTPUT "${BLDS_EMBED_CPP}" @ONLY CONTENT
"// Generated by CMakeLists.txt - do not edit.
#include \"BldsEmbedded.h\"

#if defined (__APPLE__)
 #define BLDS_SECTION       \".const_data\\n\"
 #define BLDS_LABEL(name)   \"_\" name
 #define BLDS_HIDDEN(name)  \".globl _\" name \"\\n.private_extern _\" name
#else
 #define BLDS_SECTION       \".section .rodata\\n\"
 #define BLDS_LABEL(name)   name
 #define BLDS_HIDDEN(name)  \".globl \" name \"\\n.hidden \" name
#endif

@BLDS_ASM@@BLDS_DECLS@
namespace blds
{
    static const EmbeddedFile files[] =
    {
@BLDS_ENTRIES@    };

    const EmbeddedFile* embeddedFiles() noexcept  { return files; }
    int numEmbeddedFiles() noexcept               { return @BLDS_COUNT@; }
}
")

# Rebuild the generated file when a .wav changes (the .incbin input).
set_source_files_properties("${BLDS_EMBED_CPP}" PROPERTIES OBJECT_DEPENDS "${BLDS_EMBEDDED_FILES}")

#-------------------------------------------------------------------------------
# Helper that creates one plugin variant from the shared sources.
#-------------------------------------------------------------------------------
function(blds_add_variant target product code is_effect)
    if(is_effect)
        set(main_type kAudioUnitType_Effect)
        set(vst3_cat Fx Generator)
        set(synth FALSE)
    else()
        set(main_type kAudioUnitType_MusicDevice)
        set(vst3_cat Instrument Generator)
        set(synth TRUE)
    endif()

    juce_add_plugin(${target}
        PRODUCT_NAME                "${product}"
        COMPANY_NAME                "Lnce"
        BUNDLE_ID                   "com.lnce.${target}"
        PLUGIN_MANUFACTURER_CODE    Lnce
        PLUGIN_CODE                 ${code}
        FORMATS                     ${BLDS_FORMATS}
        IS_SYNTH                    ${synth}
        NEEDS_MIDI_INPUT            FALSE
        NEEDS_MIDI_OUTPUT           FALSE
        IS_MIDI_EFFECT              FALSE
        EDITOR_WANTS_KEYBOARD_FOCUS FALSE
        AU_MAIN_TYPE                ${main_type}
        VST3_CATEGORIES             ${vst3_cat}
        HARDENED_RUNTIME_ENABLED    TRUE
        COPY_PLUGIN_AFTER_BUILD     FALSE)

    target_sources(${target} PRIVATE ${BLDS_SOURCES})
    target_include_directories(${target} PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}/Source")

    target_compile_definitions(${target}
        PUBLIC
            BLDS_EFFECT=$<BOOL:${is_effect}>
            JUCE_WEB_BROWSER=0
            JUCE_USE_CURL=0
            JUCE_VST3_CAN_REPLACE_VST2=0)

    target_link_libraries(${target}
        PUBLIC
            juce::juce_audio_utils
            juce::juce_audio_processors
            juce::juce_gui_basics
            juce::juce_recommended_config_flags
            juce::juce_recommended_warning_flags)

    if(BLDS_ENABLE_LTO)
        target_link_libraries(${target} PUBLIC juce::juce_recommended_lto_flags)
    endif()
endfunction()

blds_add_variant(BLDSPlayer   "BLDS Player"    Blds FALSE)
blds_add_variant(BLDSPlayerFX "BLDS Player FX" Blde TRUE)

#-------------------------------------------------------------------------------
# Command-line verification (no GUI, no host). One binary per variant, each
# linked against that variant's real shared plugin code.
#-------------------------------------------------------------------------------
enable_testing()

foreach(variant BLDSPlayer BLDSPlayerFX)
    add_executable(${variant}Tests Tests/BldsTests.cpp)
    target_link_libraries(${variant}Tests PRIVATE ${variant}
                          juce::juce_recommended_config_flags
                          juce::juce_recommended_warning_flags)
    target_compile_definitions(${variant}Tests PRIVATE BLDS_FILES_DIR="${BLDS_DIR}")
    add_test(NAME ${variant}Tests COMMAND ${variant}Tests)
endforeach()
